Real-Life Troubleshooting Scenarios

01 Container won't start
# Find dead containers
docker ps -a --filter "status=exited"

# Check error reason
docker inspect broken_container

# See last 50 log lines
docker logs broken_container --tail 50
02 High CPU usage
# Live CPU/memory for all containers
docker stats

# One-time snapshot
docker stats --no-stream app

# Processes inside container
docker top app
03 Disk full
# Clean everything unused
docker system prune -a

# Remove all images
docker images -q | xargs docker rmi

# Remove orphan volumes
docker volume prune
04 Port conflict
# Find containers using host network
docker ps --filter "network=host"

# Get container IP address
docker inspect app --format '{{.NetworkSettings.IPAddress}}'
05 Deploy to production
# Build with version tag
docker build -t myapp:v1.2 .

# Tag for registry
docker tag myapp:v1.2 registry.com/myapp:v1.2

# Push to registry
docker push registry.com/myapp:v1.2

# Deploy
docker run -d --name app -p 8080:80 registry.com/myapp:v1.2
